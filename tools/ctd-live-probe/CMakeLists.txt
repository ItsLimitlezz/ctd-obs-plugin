# ctd-live-probe on its own, without OBS: the engine plus main.c against the system FFmpeg (pkg-config).
# The test harness builds it this way (test/Dockerfile). Inside the plugin build, use -DENABLE_PROBE=ON instead.
cmake_minimum_required(VERSION 3.22)
project(ctd-live-probe C)

set(CMAKE_C_STANDARD 11)
find_package(PkgConfig REQUIRED)
pkg_check_modules(AVFORMAT REQUIRED IMPORTED_TARGET libavformat)
pkg_check_modules(AVCODEC REQUIRED IMPORTED_TARGET libavcodec)
pkg_check_modules(AVUTIL REQUIRED IMPORTED_TARGET libavutil)
pkg_check_modules(SWRESAMPLE REQUIRED IMPORTED_TARGET libswresample)
find_package(Threads REQUIRED)

set(ENGINE "${CMAKE_CURRENT_SOURCE_DIR}/../../src/engine")
add_executable(ctd-live-probe main.c group.c)
target_sources(ctd-live-probe PRIVATE "${ENGINE}/engine.c" "${ENGINE}/audio-norm.c" "${ENGINE}/hw-decode.c")
target_sources(ctd-live-probe PRIVATE "${ENGINE}/delay-queue.c" "${ENGINE}/ltc.c" "${ENGINE}/sync-group.c")
target_include_directories(ctd-live-probe PRIVATE "${ENGINE}")
target_link_libraries(ctd-live-probe PRIVATE PkgConfig::AVFORMAT PkgConfig::AVCODEC PkgConfig::AVUTIL)
target_link_libraries(ctd-live-probe PRIVATE PkgConfig::SWRESAMPLE Threads::Threads)
if(NOT WIN32)
  target_link_libraries(ctd-live-probe PRIVATE m)
endif()
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(ctd-live-probe PRIVATE -Wall -Wextra)
endif()
# ltc-gen: the harness's LTC source (T17)
add_executable(ltc-gen ltc-gen.c "${ENGINE}/ltc.c")
target_include_directories(ltc-gen PRIVATE "${ENGINE}")
if(NOT WIN32)
  target_link_libraries(ltc-gen PRIVATE m)
endif()
install(TARGETS ctd-live-probe ltc-gen)
